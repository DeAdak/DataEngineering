To execute PySpark .py files stored in a Google Cloud Storage (GCS) bucket, you will use the DataprocSubmitJobOperator. 
This operator instructs Google Cloud Dataproc to pull your script directly from GCS and run it on a managed Spark cluster.

Prerequisites
Before deploying this DAG, ensure you have configured your environment:
Airflow Connection: A Google Cloud connection configured in Airflow named google_cloud_default.
Dataproc Cluster: An active Dataproc cluster running in your GCP project.
File Paths: Your PySpark code must be uploaded to a bucket path (e.g., gs://your-bucket-name/scripts/my_spark_job.py).

0. Library Imports

import airflow
from airflow import DAG
from datetime import timedelta
from airflow.providers.google.cloud.operators.dataproc import (
    DataprocStartClusterOperator,
	DataprocSubmitJobOperator,
    DataprocStopClusterOperator
)

# 1. Global Configurations

PROJECT_ID = "avd-databricks-demo"
REGION = "us-east1"
CLUSTER_NAME = "my-demo-cluster"
COMPOSER_BUCKET = "us-central1-demo-instance-a649b781-bucket"

# Define path to your primary PySpark script and any helper files inside GCS

hospitalALanding_URI = f'gs://{COMPOSER_BUCKET}/data/INGESTION/hospitalALanding.py"
hospitalBLanding_URI = f'gs://{COMPOSER_BUCKET}/data/INGESTION/hospitalBLanding.py"
claimsLanding_URI = f'gs://{COMPOSER_BUCKET}/data/INGESTION/hospitalBLanding.py"

# 2. Default Task Configurations
default_args = {
    "owner": "DEBABRATA ADAK",
    "start_date": datetime(2026, 1, 1),
	#Airflow requires a fixed, concrete static date (like datetime(2026, 1, 1)) to initialize a DAG. Setting it to None causes a crash.
    "depends_on_past": False,
    "email_on_failure": False,
    "email_on_retry": False,
    "email": ["***@gmail.com"],
    "email_on_success": False,
    "retries": 1,
    "retry_delay": timedelta(minutes=5)
}

# 3. DAG Definition
with DAG(
    dag_id='gcs_pyspark_dataproc_pipeline', # Use in parent DAG
    default_args=default_args,
    description='Submits a PySpark script from a GCS bucket to a Dataproc cluster',
    schedule_interval=None, #Parent DAG will handle the scheduling.
    #start_date=datetime(2026, 1, 1),
    catchup=False,
    tags=['gcp', 'dataproc', 'pyspark'],
) as dag:

# 4. Define the Dataproc PySpark Job Structural Payload

hospitalALandingPYSPARK_JOB_1 = {
    "reference": {"project_id": PROJECT_ID},
    "placement": {"cluster_name": CLUSTER_NAME},
    "pyspark_job": {"main_python_file_uri": hospitalALanding_URI}
}

hospitalBLandingPYSPARK_JOB_2 = {
    "reference": {"project_id": PROJECT_ID},
    "placement": {"cluster_name": CLUSTER_NAME},
    "pyspark_job": {"main_python_file_uri": hospitalBLanding_URI}
}

claimsLandingPYSPARK_JOB_3 = {
	"reference": {"project_id": PROJECT_ID},
    "placement": {"cluster_name": CLUSTER_NAME},
    "pyspark_job": {"main_python_file_uri": claimsLanding_URI}
}

# 5. Task Instantiation
start_cluster = DataprocStartClusterOperator(
	task_id = "start_cluster",
	project_id = PROJECT_ID,
	region = REGION,
	cluster_name = CLUSTER_NAME
)

hospitalALanding = DataprocSubmitJobOperator(
	task_id="hospitalALanding", 
	job=hospitalALandingPYSPARK_JOB_1, 
	region=REGION, 
	project_id=PROJECT_ID
)

hospitalBLanding = DataprocSubmitJobOperator(
	task_id="hospitalBLanding", 
	job=hospitalALandingPYSPARK_JOB_2, 
	region=REGION, 
	project_id=PROJECT_ID
)

claimsLanding = DataprocSubmitJobOperator(
	task_id="claimsLanding", 
	job=claimsLandingPYSPARK_JOB_3, 
	region=REGION, 
	project_id=PROJECT_ID
)

stop_cluster = DataprocStopClusterOperator(
	task_id = "stop_cluster",
	project_id = PROJECT_ID,
	region = REGION,
	cluster_name = CLUSTER_NAME
	)
