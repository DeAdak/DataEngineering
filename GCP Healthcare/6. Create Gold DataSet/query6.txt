Step 1: Create the Gold Target Table (DDL)
CREATE OR REPLACE TABLE `avd-databricks-demo.gold_dataset.payor_performance` (
    PayorID             	STRING    	OPTIONS(description="The unique identifier for the insurance payor (e.g. Medicare, BlueCross)"),
    PayorType           	STRING    	OPTIONS(description="Classification category (e.g. Private, Government, Self-pay)"),
    TotalClaimsSubmitted 	INT64    	OPTIONS(description="Total distinct claims sent to this payor"),
    TotalClaimsApproved  	INT64    	OPTIONS(description="Total distinct claims approved or paid by this payor"),
    ClaimApprovalRate   	FLOAT64   	OPTIONS(description="Percentage of submitted claims that were approved/paid"),
    TotalClaimedAmount  	NUMERIC   	OPTIONS(description="The total gross money billed to the insurance payor"),
    TotalPaidAmount     	NUMERIC   	OPTIONS(description="The actual monetary payout collected from the payor"),
    AverageProcessingDays 	FLOAT64 	OPTIONS(description="Average days elapsed between Service Date and Claim Submission Date")
);

##------------------------------------------------------------------------------------------------------------------------##
Step 2: Extract, Clean, and Populate the Table (DML)
INSERT INTO `avd-databricks-demo.gold_dataset.payor_performance`

WITH verified_claims_data AS (
    SELECT 
        PayorID,
        PayorType,
        SRC_ClaimID,
        ClaimStatus,
        CAST(ClaimAmount AS NUMERIC) AS ClaimAmount,
        CAST(PaidAmount AS NUMERIC) AS PaidAmount,
        -- Safe date parsing to convert 'DD-MM-YYYY' string attributes into functional dates
        SAFE.PARSE_DATE('%d-%m-%Y', ServiceDate) AS parsed_service_date,
        SAFE.PARSE_DATE('%d-%m-%Y', ClaimDate) AS parsed_claim_date
    FROM `avd-databricks-demo.silver_dataset.claims`
    WHERE is_quarantined = FALSE 
      AND PayorID IS NOT NULL
)

SELECT 
    PayorID,
    MAX(PayorType) AS PayorType, -- Aggregate fallback grouping strategy
    COUNT(DISTINCT SRC_ClaimID) AS TotalClaimsSubmitted,
    COUNT(DISTINCT CASE WHEN ClaimStatus IN ('Approved', 'Paid') THEN SRC_ClaimID END) AS TotalClaimsApproved,
    
    -- 1. Calculate Claim Approval Rate
    ROUND(
        COUNT(DISTINCT CASE WHEN ClaimStatus IN ('Approved', 'Paid') THEN SRC_ClaimID END) / 
        NULLIF(COUNT(DISTINCT SRC_ClaimID), 0) * 100, 2
    ) AS ClaimApprovalRate,
    
    -- 2. Aggregate Financial Payout Totals
    SUM(ClaimAmount) AS TotalClaimedAmount,
    SUM(PaidAmount) AS TotalPaidAmount,
    
    -- 3. Calculate Processing Efficiency (Average Days Elapsed)
    ROUND(
        AVG(DATE_DIFF(parsed_claim_date, parsed_service_date, DAY)), 1
    ) AS AverageProcessingDays

FROM verified_claims_data
GROUP BY PayorID;
##------------------------------------------------------------------------------------------------------------------------##
Key Safeguards inside this Architecture:
Safe Date Handling: Using SAFE.PARSE_DATE prevents the entire data warehouse pipeline from crashing if a row contains an invalid date value like 31-02-2024.
Accurate Averages: The AVG(DATE_DIFF(...)) metric computes processing speeds directly from granular rows, providing an accurate look at operational bottlenecks.
No Fan-Out Duplication: The script aggregates data entirely within the boundaries of the claims table to prevent table duplication or inflated currency figures.
