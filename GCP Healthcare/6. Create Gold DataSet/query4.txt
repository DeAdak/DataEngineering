---Provides insights into department-level efficiency, revenue, and patient volume.
INSERT IN `avd-databricks-demo.gold_dataset.department_performance`
SELECT 	d.SRC_DeptID AS DepartmentID,
		d.Name AS DepartmentName,
		COUNT(DISTINCT e.EncounterID) AS TotalEncounters,
		COUNT(DISTINCT t.SRC_TransactionID) AS TotalTransactions,
		SUM(t.Amount) AS TotalBilledAmount,
		SUM(t.PaidAmount) AS TotalPaidAmount,
		COUNT(DISTINCT e.PatientID) AS UniquePatients,
		COUNT(DISTINCT e.ProviderID) AS ActiveProviders,
		ROUND((SUM(CAST (t.PaidAmount AS NUMERIC))/NULLIF(SUM(CAST((t.Amount AS NUMERIC))),0)*100,2) AS RevenueCollectionRate 
FROM `avd-databricks-demo.silver_dataset.departments` d
LEFT JOIN
`avd-databricks-demo.silver_dataset.encounters` e ON d.SRC_DeptID = e.DepartmentID
LEFT JOIN 
`avd-databricks-demo.silver_dataset.transactions` t ON d.SRC_DeptID = t.DeptID
WHERE is_quarantined = FALSE
GROUP BY d.SRC_DeptID, d.Name;

CREATE TABLE IF NOT EXISTS `avd-databricks-demo.gold_dataset.department_performance`(
		DepartmentID			STRING,
		DepartmentName			STRING,
		TotalEncounters			INT,
		TotalTransactions		INT,
		TotalBilledAmount		NUMERIC,
		TotalPaidAmount			NUMERIC,
		UniquePatients			INT,
		ActiveProviders			INT,
		RevenueCollectionRate FLOAT64
		);

TRUNCATE TABLE `avd-databricks-demo.gold_dataset.department_performance`;
