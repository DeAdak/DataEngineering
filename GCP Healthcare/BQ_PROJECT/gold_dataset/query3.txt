---This table summarizes provider activity, including the number of encounters/doctor, total billed amount/doctor, and claim success rate/doctor.
INSERT INTO `avd-databricks-demo.gold_dataset.provider_performance`
SELECT 
		p.ProviderID,
		p.FirstName,
		p.LastName,
		p.Specialization,
		COUNT(DISTINCT e.EncounterID) AS TotalEncounters,
		COUNT(DISTINCT t.SRC_TransactionID) AS TotalTransactions,
		SUM(t.Amount) AS TotalBilledAmount,
		SUM(t.PaidAmount) AS TotalPaidAmount,
		COUNT(DISTINCT c.SRC_ClaimID) AS TotalClaims,
		COUNT(CASE WHEN c.ClaimStatus = 'Approved' THEN SRC_ClaimID END) AS ApprovedClaims
		ROUND(
				COUNT(CASE WHEN c.ClaimStatus = 'Approved' THEN SRC_ClaimID END) / NULLIF(COUNT(DISTINCT c.SRC_ClaimID),0) * 100 ,2		
			) AS ClaimSuccessRate
FROM
avd-databricks-demo.silver_dataset.providers p
LEFT JOIN
avd-databricks-demo.silver_dataset.encounters e ON SPLIT(p.ProviderID,'-')[SAFE_OFFSET(1)] = e.ProviderID
LEFT JOIN
avd-databricks-demo.silver_dataset.transactions t ON SPLIT(p.ProviderID,'-')[SAFE_OFFSET(1)] = t.ProviderID
LEFT JOIN
avd-databricks-demo.silver_dataset.claims c ON SPLIT(p.ProviderID,'-')[SAFE_OFFSET(1)] = C.ProviderID
GROUP BY p.ProviderID, p.FirstName,	p.LastName,	p.Specialization;

CREATE TABLE `avd-databricks-demo.gold_dataset.provider_performance`(
				ProviderID			STRING,
                FirstName			STRING,
                LastName			STRING,
                Specialization		STRING,
                TotalEncounters 	INT64,
                TotalTransactions	INT64,
                TotalBilledAmount	NUMERIC,
                TotalPaidAmount		NUMERIC,
                TotalClaims			INT64,
				ApprovedClaims		INT64,
                ClaimSuccessRate	FLOAT64
);

TRUNCATE TABLE `avd-databricks-demo.gold_dataset.provider_performance`;
