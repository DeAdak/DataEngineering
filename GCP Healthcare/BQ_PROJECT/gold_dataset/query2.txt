---4. Insert data into `avd-databricks-demo.gold_dataset.patient_history` table
INSERT INTO `avd-databricks-demo.gold_dataset.patient_history`
---1. Select all the required Columns from silver dataSet
SELECT 
		p.Patient_Key,
		p.FirstName,
		p.LastName,
		p.MiddleName,
		p.Gender,
		p.DOB,
		p.Address,
		e.EncounterDate,
		e.EncounterType,
		CONCAT(pr.FirstName,' ',pr.LastName) as Doctor
		t.SRC_TransactionID
		t.VisitDate
		t.ServiceDate
		t.PaidDate
		t.Amount AS BilledAmount
		c.PaidAmount 
		c.ClaimAmount
		c.PaidAmount AS ClaimPaidAmount
		c.ClaimStatus
		c.PayorType

FROM 
avd-databricks-demo.silver_dataset.patients p
LEFT JOIN
avd-databricks-demo.silver_dataset.encounters e ON p.SRC_PatientID = e.PatientID
LEFT JOIN
avd-databricks-demo.silver_dataset.providers pr ON e.ProviderID = SPLIT(pr.ProviderID,'-')[OFFSET(1)]
LEFT JOIN  
avd-databricks-demo.silver_dataset.transactions t ON p.SRC_PatientID = t.PatientID
LEFT JOIN
avd-databricks-demo.silver_dataset.claims c ON t.SRC_TransactionID = c.TransactionID
WHERE is_current = TRUE;

---2. Create the table to put the required data
CREATE TABLE IF NOT EXISTS `avd-databricks-demo.gold_dataset.patient_history`(
							Patient_Key		STRING,
                            FirstName          STRING,
                            LastName           STRING,
                            MiddleName         STRING,
                            Gender             STRING,
                            DOB                DATE,
                            Address            STRING,
                            EncounterDate      DATE,
                            EncounterType      DATE,
                            Doctor             STRING,
                            TransactionID	   STRING,
                            VisitDate          DATE,
                            ServiceDate        DATE,
                            PaidDate           DATE,
                            BilledAmount       NUMERIC,
                            PaidAmount         NUMERIC,
                            ClaimAmount        NUMERIC,
                            ClaimPaidAmount    NUMERIC,
                            ClaimStatus        STRING,
                            PayorType          STRING
);

---3. Truncate to remove old values(if exist)
TRUNCATE TABLE `avd-databricks-demo.gold_dataset.patient_history`;
