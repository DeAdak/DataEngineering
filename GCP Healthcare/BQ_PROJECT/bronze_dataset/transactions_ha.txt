CREATE EXTERNAL TABLE IF NOT EXISTS `avd-databricks-demo.bronze_dataset.transactions_ha` 
OPTIONS (
  format = 'JSON',
  uris = ['gs://healthcare-bucket-22032025/landing/HospitalA/transactions/*.json']
);

##--------------------------------------------------##
1. Explicit Strict Schema: 
Mapping fields directly to STRING, DATE, and NUMERIC guarantees that any underlying application query performs deterministically and eliminates data 
type inference overhead.

2. Financial Data Precision (NUMERIC): 
Medical billing and transaction systems should never use FLOAT64 (double precision) due to binary rounding errors. 
BigQuery's NUMERIC type maintains exact decimal precision.

3. Resilience with ignore_unknown_values = true: 
If an application update pushes a new JSON key-value pair into GCS that isn't explicitly defined in your schema, BigQuery safely ignores it instead 
of completely failing your analytical queries.

4. Data Quality with max_bad_records = 0: 
This ensures that if a row is completely corrupted (e.g., malformed JSON), BigQuery throws an alert rather than silently skipping the row.

CREATE EXTERNAL TABLE IF NOT EXISTS `avd-databricks-demo.bronze_dataset.transactions_ha` (
  TransactionID STRING,
  EncounterID STRING,
  PatientID STRING,
  ProviderID STRING,
  DeptID STRING,
  VisitDate DATE,
  ServiceDate DATE,
  PaidDate DATE,
  VisitType STRING,
  Amount NUMERIC,      -- Use NUMERIC instead of FLOAT64 for precise financial data
  AmountType STRING,
  PaidAmount NUMERIC,  -- Use NUMERIC instead of FLOAT64 for precise financial data
  ClaimID STRING,
  PayorID STRING,
  ProcedureCode STRING,
  ICDCode STRING,
  LineOfBusiness STRING,
  MedicaidID STRING,
  MedicareID STRING,
  InsertDate DATE,
  ModifiedDate DATE
)
OPTIONS (
  format = 'JSON',
  uris = ['gs://healthcare-bucket-22032025/landing/HospitalA/transactions/*.json'],
  ignore_unknown_values = true, -- Prevents query crashes if upstream apps add new fields
  max_bad_records = 0          -- Enforces strict data quality compliance
);
