CREATE OR  REPLACE TABLE `avd-databricks-demo.silver_dataset.quality_checks` AS
SELECT DISTINCT 
	CONCAT(EncounterID,'-',datasource) AS Encounter_Key,
	EncounterID,
	PatientID,
	EncounterDate,
	EncounterType,
	ProviderID,
	DepartmentID,
	ProcedureCode,
	InsertedDate as SRC_InsertedDate,
	ModifiedDate as SRC_ModifiedDate,
	datasource, 
	CASE WHEN SRC_EncounterID IS NULL OR PatientID IS NULL OR EncounterDate IS NULL OR LOWER(EncounterType) = 'null' THEN TRUE
        ELSE FALSE
    END AS is_quarantined

FROM(
	SELECT DISTINCT *, 'HospitalA' AS datasource FROM `avd-databricks-demo.bronze_dataset.encounters_ha`
	UNION ALL
	SELECT DISTINCT *, 'HospitalB' AS datasource FROM `avd-databricks-demo.bronze_dataset.encounters_hb`
	)

CREATE TABLE IF NOT EXISTS `avd-databricks-demo.gold_dataset.encounters`(
							Encounter_Key		STRING,
							EncounterID         STRING,
							PatientID           STRING,
							EncounterDate       DATE,
							EncounterType       STRING,
							ProviderID          STRING,
							DepartmentID        STRING,
							ProcedureCode       INT64,
							SRC_InsertedDate	DATE,
							SRC_ModifiedDate	DATE,
							datasource	        STRING,
							is_qurantined	    BOOL,
							inserted_date	    DATE,
							modified_date	    DATE,
							is_current	        BOOL
							)
							
MERGE INTO `avd-databricks-demo.gold_dataset.encounters` as target
USING `avd-databricks-demo.silver_dataset.quality_checks` as source
ON target.Encounter_Key = source.Encounter_Key AND is_current = TRUE
WHEN MATCHHED AND (
                    source.EncounterID         <> target.EncounterID,
                    source.PatientID           <> target.PatientID,
                    source.EncounterDate       <> target.EncounterDate,
                    source.EncounterType       <> target.EncounterType,
                    source.ProviderID          <> target.ProviderID,
                    source.DepartmentID        <> target.DepartmentID,
                    source.ProcedureCode       <> target.ProcedureCode,
                    source.SRC_InsertedDate    <> target.SRC_InsertedDate,
                    source.SRC_ModifiedDate    <> target.SRC_ModifiedDate,
                    source.datasource          <> target.datasource,
                    source.is_qurantined       <> target.is_qurantined
					)
THEN UPDATE target.is_current = FALSE,
target.modified_date = CURRENT_TIMESTAMP()
WHEN NOT MATCHED INSERT (
						Encounter_Key,
                        EncounterID,
                        PatientID,
                        EncounterDate,
                        EncounterType,
                        ProviderID,
                        DepartmentID,
                        ProcedureCode,
                        SRC_InsertedDate,
                        SRC_ModifiedDate,
                        datasource,
                        is_qurantined,
                        inserted_date,
                        modified_date,
                        is_current
						) VALUES (
						source.Encounter_Key,
						source.EncounterID,
						source.PatientID,
						source.EncounterDate,
						source.EncounterType,
						source.ProviderID,
						source.DepartmentID,
						source.ProcedureCode,
						source.SRC_InsertedDate,
						source.SRC_ModifiedDate,
						source.datasource,
						source.is_qurantined,
						CURRENT_TIMESTAMP(),
						CURRENT_TIMESTAMP(),
						TRUE						
						);

DROP TABLE IF EXISTS `avd-databricks-demo.silver_dataset.quality_checks`;
