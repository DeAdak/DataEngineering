quality_checks
-------------
CREATE OR REPLACE TABLE `avd-databricks-demo.silver_dataset.quality_checks` AS
SELECT DISTINCT CONCAT(ClaimID,'-',datasource) as Claim_Key,
				ClaimID AS SRC_ClaimID,
				TransactionID,
				PatientID,
				EncounterID,
				ProviderID,
				DeptID,
				ServiceDate,
				ClaimDate,
				PayorID,
				ClaimAmount,
				PaidAmount,
				ClaimStatus,
				PayorType,
				Deductible,
				Coinsurance,
				Copay,
				InsertDate AS SRC_InsertDate,
				ModifiedDate AS SRC_ModifiedDate,
				datasource,
				CASE WHEN SRC_ClaimID IS NULL OR PatientID IS NULL OR TransactionID IS NULL OR LOWER(ClaimStatus) = 'null' THEN TRUE ELSE END AS is_qurantined
FROM `avd-databricks-demo.bronze_dataset.claims`

avd-databricks-demo.silver_dataset.claims
------------------------------------------
CREATE TABLE IF NOT EXISTS `avd-databricks-demo.silver_dataset.claims`(
							Claim_Key		  STRING,
                            SRC_ClaimID       STRING,
                            TransactionID     STRING,
                            PatientID         STRING,
                            EncounterID       STRING,
                            ProviderID        STRING,
                            DeptID            STRING,
                            ServiceDate       DATE,
                            ClaimDate         DATE,
                            PayorID           STRING,
                            ClaimAmount       NUMERIC,
                            PaidAmount        NUMERIC,
                            ClaimStatus       STRING,
                            PayorType         STRING,
                            Deductible        NUMERIC,
                            Coinsurance       NUMERIC,
                            Copay             NUMERIC,
                            SRC_InsertDate    DATE,
                            SRC_ModifiedDate  DATE,
                            datasource        STRING,
							is_qurantined     BOOL,
							inserted_date	  DATE,
							modified_date	  DATE,
							is_current		  BOOL
							);
							
MERGE INTO `avd-databricks-demo.silver_dataset.claims` AS target
USING 'avd-databricks-demo.silver_dataset.quality_checks' AS source
ON target.is_current = TRUE AND target.Claim_Key = source.Claim_Key
WHEN MATCHED AND (		
					source.SRC_ClaimID     <> target.SRC_ClaimID,     
					source.TransactionID   <> target.TransactionID,   
					source.PatientID       <> target.PatientID,       
					source.EncounterID     <> target.EncounterID,     
					source.ProviderID      <> target.ProviderID,      
					source.DeptID          <> target.DeptID,          
					source.ServiceDate     <> target.ServiceDate,     
					source.ClaimDate       <> target.ClaimDate,       
                    source.PayorID         <> target.PayorID,         
                    source.ClaimAmount     <> target.ClaimAmount,     
                    source.PaidAmount      <> target.PaidAmount,      
                    source.ClaimStatus     <> target.ClaimStatus,     
                    source.PayorType       <> target.PayorType,       
                    source.Deductible      <> target.Deductible,      
                    source.Coinsurance     <> target.Coinsurance,     
                    source.Copay           <> target.Copay,           
                    source.SRC_InsertDate  <> target.SRC_InsertDate,  
                    source.SRC_ModifiedDate<> target.SRC_ModifiedDate,
                    source.datasource      <> target.datasource,      
                    source.is_qurantined   <> target.is_qurantined
) THEN
UPDATE SET target.is_current = FALSE, target.modified_date = CURRENT_TIMESTAMP()

WHEN NOT MATCHED THEN INSERT(
							Claim_Key,		
							SRC_ClaimID,     
                            TransactionID,  
                            PatientID,       
                            EncounterID,     
                            ProviderID,      
                            DeptID,          
                            ServiceDate,     
                            ClaimDate,       
                            PayorID,         
                            ClaimAmount,     
                            PaidAmount,      
                            ClaimStatus,     
                            PayorType,       
                            Deductible,      
                            Coinsurance,     
                            Copay,           
                            SRC_InsertDate,  
                            SRC_ModifiedDate,
                            datasource,      
                            is_qurantined,   
                            inserted_date,	
                            modified_date,	
                            is_current
							)
VALUES 						(
							source.Claim_Key,		
                            source.SRC_ClaimID,     
                            source.TransactionID,   
                            source.PatientID,       
                            source.EncounterID,     
                            source.ProviderID,      
                            source.DeptID,          
                            source.ServiceDate,     
                            source.ClaimDate,       
                            source.PayorID,         
                            source.ClaimAmount,     
                            source.PaidAmount,      
                            source.ClaimStatus,     
                            source.PayorType,       
                            source.Deductible,      
                            source.Coinsurance,     
                            source.Copay,           
                            source.SRC_InsertDate,  
                            source.SRC_ModifiedDate,
                            source.datasource,      
                            source.is_qurantined,
							CURRENT_TIMESTAMP(),
							CURRENT_TIMESTAMP(),
							TRUE
);

DROP TABLE avd-databricks-demo.silver_dataset.quality_checks;
