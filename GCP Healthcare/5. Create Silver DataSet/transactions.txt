	Transaction_Key		STRING,

    DeptID              STRING,
    VisitDate           DATE,
    ServiceDate         DATE,

    SRC_InsertDate      DATE,
    SRC_ModifiedDate    DATE,
    datasource          STRING,
    is_quarantine       BOOL,
    inserted_date       TIMESTAMP,
    modified_date       TIMESTAMP,
    is_current          BOOL

	PARTITION BY DATE(inserted_date)
