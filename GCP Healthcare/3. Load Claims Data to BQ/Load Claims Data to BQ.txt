from pyspark.sql import SparkSession
from pyspark.sql.functions import input_file_name,when,contains


spark = SparkSession.builder.appName('Claims').getOrCreate()

# Google Cloud Storage (GCS) Configuration
GCS_BUCKET = 'healthcare-bucket-22032025'
CLAIMS_FILE_PATH = f'gs://{GCS_BUCKET}/landing/claims/*.csv'
TEMP_GCS_BUCKET = f'gs://{GCS_BUCKET}/temp/'

#BigQuery Configuration
BQ_PROJECT = 'avd-databricks-demo'
BQ_CLAIMS_TABLE = f'{BQ_PROJECT}.bronze_dataset.claims'

By default, spark.read.csv reads all columns as STRING data types unless you explicitly define a schema or pass .option("inferSchema", "true").
claims_df = (spark.read.option('inferSchema','true').csv(CLAIMS_FILE_PATH ,header = True))

claims_df = claims_df.withColumn('datasource',
					when(contains(input_file_name(),'hospital1'),'HospitalA')
					.when(contains(input_file_name(),'hospital2'),'HospitalB')
					.otherwise('None')))
					
claims_df = claims_df.dropDuplicates()

# inferSchema is not a valid option for writing data.inferSchema is an option used exclusively when reading raw files (like CSVs or JSON) to help Spark guess the data types. When you are writing a
 DataFrame to BigQuery, Spark already knows what types its columns are.
(claims_df.write.format('bigquery')
				.option('table',BQ_CLAIMS_TABLE)
				.option('temporaryGcsBucket',TEMP_GCS_BUCKET)
				.mode('append')
				.save())
